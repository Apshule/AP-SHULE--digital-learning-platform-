    <script>
        // ==========================================
        // 1. CLOCK
        // ==========================================
        function updateClock() {
            const now = new Date();
            const timeString = now.toLocaleTimeString('en-GB', { timeZone: 'Africa/Kampala' });
            const el = document.getElementById('clock');
            if (el) el.textContent = timeString;
        }
        setInterval(updateClock, 1000);
        updateClock();

        // ==========================================
        // 2. LOGIN & ROLE ROUTING
        // ==========================================
        function handleLogin() {
            const email = document.getElementById('login-email').value.toLowerCase();
            const type = document.getElementById('login-account-type').value;
            if (!email) { alert('Please enter an email address.'); return; }
            
            document.getElementById('login-view').style.display = 'none';
            document.getElementById('app-view').style.display = 'flex';
            
            // Route based on email keyword OR dropdown type
            if (email.includes('bursar') || type === 'bursar') {
                loadRole('bursar');
            } else if (email.includes('secondary') || email.includes('headteacher') || type === 'secondary') {
                loadRole('secondary');
            } else {
                loadRole('primary');
            }
        }

        function handleLogout() {
            if (confirm('Are you sure you want to log out?')) {
                document.getElementById('app-view').style.display = 'none';
                document.getElementById('login-view').style.display = 'flex';
                document.getElementById('login-email').value = '';
                document.getElementById('login-password').value = '';
            }
        }

        // ==========================================
        // 3. ROLE-BASED CONTENT LOADER
        // ==========================================
        function loadRole(role) {
            const sidebar = document.getElementById('main-menu');
            const contentArea = document.getElementById('main-content-area');
            const roleBadge = document.getElementById('navbar-role');
            const schoolName = document.getElementById('navbar-school-name');

            if (role === 'primary') {
                roleBadge.textContent = 'Secretary';
                schoolName.textContent = 'Apshule Primary School';
                sidebar.innerHTML = getPrimarySidebar();
                contentArea.innerHTML = getPrimaryViews();
                navigateTo('primary-dashboard');
            } 
            else if (role === 'secondary') {
                roleBadge.textContent = 'Headteacher';
                schoolName.textContent = 'Apshule Secondary School';
                sidebar.innerHTML = getSecondarySidebar();
                contentArea.innerHTML = getSecondaryViews();
                navigateTo('sec-dashboard');
            }
            else if (role === 'bursar') {
                roleBadge.textContent = 'Bursar';
                schoolName.textContent = 'Apshule Finance Department';
                sidebar.innerHTML = getBursarSidebar();
                contentArea.innerHTML = getBursarViews();
                navigateTo('bursar-dashboard');
            }
        }

        // ==========================================
        // 4. NAVIGATION & SUBMENU LOGIC
        // ==========================================
        function navigateTo(viewName) {
            document.querySelectorAll('.view-section').forEach(sec => sec.classList.remove('active'));
            const target = document.getElementById('view-' + viewName);
            if (target) { 
                target.classList.add('active'); 
            } else { 
                alert('Module under construction: ' + viewName); 
                return; 
            }

            document.querySelectorAll('.sidebar-menu > li').forEach(item => item.classList.remove('active'));
            document.querySelectorAll('.sidebar-menu > li').forEach(item => {
                if (item.textContent.trim().toLowerCase().includes(viewName.split('-')[0])) {
                    item.classList.add('active');
                }
            });
        }

        function toggleSubmenu(id) {
            const el = document.getElementById(id);
            if (el) el.classList.toggle('open');
        }

        function toggleSidebar() {
            const sidebar = document.querySelector('.sidebar');
            if (sidebar.style.width === '70px') {
                sidebar.style.width = 'var(--sidebar-width)';
            } else {
                sidebar.style.width = '70px';
            }
        }

        // ==========================================
        // 5. MODAL & WIZARD LOGIC

        function openModal(modalId) { document.getElementById(modalId).classList.add('active'); }
        function closeModal(modalId) { document.getElementById(modalId).classList.remove('active'); }
        
        window.onclick = function(event) {
            if (event.target.classList.contains('modal-overlay')) { event.target.classList.remove('active'); }
        }

        function nextWizardStep(module, currentStep) {
            document.getElementById(module + '-panel-' + currentStep).style.display = 'none';
            const nextStep = currentStep + 1;
            document.getElementById(module + '-panel-' + nextStep).style.display = 'block';
            document.getElementById(module + '-step-ind-' + currentStep).classList.remove('active');
            document.getElementById(module + '-step-ind-' + currentStep).classList.add('completed');
            document.getElementById(module + '-step-ind-' + nextStep).classList.add('active');
        }

        function prevWizardStep(module, currentStep) {
            document.getElementById(module + '-panel-' + currentStep).style.display = 'none';
            const prevStep = currentStep - 1;
            document.getElementById(module + '-panel-' + prevStep).style.display = 'block';
            document.getElementById(module + '-step-ind-' + currentStep).classList.remove('active');
            document.getElementById(module + '-step-ind-' + prevStep).classList.add('active');
            document.getElementById(module + '-step-ind-' + prevStep).classList.remove('completed');
        }

        function switchSecTab(tabId, panelId) {
            document.querySelectorAll('.sec-tab').forEach(tab => tab.classList.remove('active'));
            document.querySelectorAll('.sec-panel').forEach(panel => panel.classList.remove('active'));
            document.getElementById(tabId).classList.add('active');
            document.getElementById(panelId).classList.add('active');
        }
